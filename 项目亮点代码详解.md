## 1. JWT 双令牌认证体系

设计 Access Token（7天）+ Refresh Token（30天）双令牌方案，实现用户无感刷新 Token。

### JWT 生成与校验 — `config/security/JwtProvider.java`

```java
// 双令牌通过不同 subject 区分，防止混用
public String generateAccessToken(String userId) {
    return Jwts.builder()
            .subject("user token")              // Access Token 标识
            .claim("userId", userId)
            .expiration(new Date(System.currentTimeMillis() + accessTokenExpireMs))
            .signWith(secretKey).compact();
}

public String generateRefreshToken(String userId) {
    return Jwts.builder()
            .subject("refresh token")           // Refresh Token 标识
            .claim("userId", userId)
            .expiration(new Date(System.currentTimeMillis() + refreshTokenExpireMs))
            .signWith(secretKey).compact();
}

// 校验时要求 subject 必须匹配，拒绝跨类型使用
public Claims parseAccessToken(String token) {
    return Jwts.parser().verifyWith(secretKey).build()
            .parseSignedClaims(token)
            .requireSubject("user token")       // Access Token 专用
            .getPayload();
}

public Claims parseRefreshToken(String token) {
    return Jwts.parser().verifyWith(secretKey).build()
            .parseSignedClaims(token)
            .requireSubject("refresh token")    // Refresh Token 专用
            .getPayload();
}
```

配置（`application.yml`）：

```yaml
jwt:
  secret: ${JWT_SECRET}   # 通过环境变量注入，不写入仓库
  access-token-expire: 7d
  refresh-token-expire: 30d
```

### 一次一用 Refresh Token — `service/impl/UserServiceImpl.java`

```java
public TokenPair refresh(RefreshTokenDTO dto) {
    // ① JWT 签名校验（subject 必须是 "refresh token"）
    Claims claims = jwtProvider.parseRefreshToken(dto.getRefreshToken());

    // ② 数据库查询，要求 is_used = 0（一次性使用）
    RefreshToken rt = refreshTokenMapper.selectByTokenAndNotUsed(dto.getRefreshToken());
    if (rt == null) throw new BizException("refresh token无效");

    // ③ 检查 DB 中的过期时间
    if (LocalDateTime.now().isAfter(rt.getExpiresAt())) {
        rt.setIsUsed(true);
        refreshTokenMapper.updateById(rt);
        throw new BizException("refresh token已过期");
    }

    // ④ 提取 userId，验证用户存在，签发全新 token 对
    Integer userId = Integer.parseInt(claims.get("userId", String.class));
    User user = userMapper.selectById(userId);
    if (user == null) throw new BizException("用户不存在");

    return issueTokenPair(user);
}

// 签发 token 对：生成 JWT + 写入 DB
private TokenPair issueTokenPair(User user) {
    String accessToken = jwtProvider.generateAccessToken(user.getUserId().toString());
    String refreshToken = jwtProvider.generateRefreshToken(user.getUserId().toString());

    RefreshToken record = new RefreshToken();
    record.setUserId(user.getUserId());
    record.setToken(refreshToken);
    record.setExpiresAt(LocalDateTime.now().plusDays(30));
    record.setIsUsed(false);
    record.setCreatedAt(LocalDateTime.now());
    refreshTokenMapper.insert(record);

    return new TokenPair(accessToken, refreshToken);
}
```

### OncePerRequestFilter 认证链 — `config/security/JwtAuthenticationFilter.java`

```java
@Component
public class JwtAuthenticationFilter extends OncePerRequestFilter {

    // 白名单端点无需认证
    private static final List<String> WHITE_LIST = List.of(
        "/api/auth/login", "/api/auth/register", "/api/auth/validateEmail",
        "/api/auth/refresh", "/api/auth/modifyPassword", "/websocket/"
    );

    @Override
    protected void doFilterInternal(HttpServletRequest request,
                                     HttpServletResponse response,
                                     FilterChain filterChain) throws ServletException, IOException {
        if (request.getMethod().equals("OPTIONS")) {
            filterChain.doFilter(request, response); return;  // CORS 预检放行
        }

        String token = resolveToken(request);  // 从 Header 或 Query 参数提取
        if (StringUtils.hasText(token)) {
            try {
                UserContext.setUserId(jwtProvider.getUserId(token));  // ThreadLocal 存储
            } catch (Exception e) {
                response.sendError(401, "token 无效"); return;
            }
        }

        String path = request.getRequestURI();
        if (isWhiteListed(path) || path.startsWith("/api/auth/")) {
            filterChain.doFilter(request, response);  // 白名单放行
        } else if (token == null) {
            response.sendError(401, "未登录");         // 非白名单无 token → 401
        } else {
            filterChain.doFilter(request, response);  // 有 token → 放行
        }

        UserContext.clear();  // finally 块清理 ThreadLocal，防止内存泄漏
    }

    // 支持从 Header 或 Query 参数提取 token
    private String resolveToken(HttpServletRequest request) {
        String bearer = request.getHeader("Authorization");
        if (StringUtils.hasText(bearer) && bearer.startsWith("Bearer ")) {
            return bearer.substring(7);
        }
        return request.getParameter("token");  // WebSocket 握手场景使用
    }
}
```

### 前端无感刷新 — `fronted/newSSE/src/api/req.ts`

```typescript
// 模块级单例 refreshPromise：多个并发 401 只发一次刷新请求
let refreshPromise: Promise<void> | null = null;

async function refreshTokenIfNeeded() {
    const { refreshToken, setToken, setRefreshToken } = useUserStore();
    if (!refreshToken.value) throw new Error('No refresh token');

    const res = await fetch(`${apiUrl}/auth/refresh`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ refreshToken: refreshToken.value }),
    });
    if (!res.ok) throw new Error('Refresh token request failed');

    const data = await res.json();
    const token = data.data?.token;
    const newRefreshToken = data.data?.refreshToken ?? data.data?.refresh_token;
    if (!token || !newRefreshToken) throw new Error('Invalid refresh response');

    setToken(token);           // 更新内存中的 access token
    setRefreshToken(newRefreshToken);  // 更新 refresh token
}

async function requestFunc(url: string, options: RequestInit, tokenIsNeeded = true) {
    const { token } = useUserStore();

    // 情况 1：有 access token，直接发请求
    if (token.value) {
        options.headers = { ...options.headers, Authorization: `Bearer ${token.value}` };
    }
    // 情况 2：无 access token 但有 refresh token，先刷新
    else if (refreshToken.value) {
        await refreshTokenIfNeeded();
        options.headers = { ...options.headers, Authorization: `Bearer ${token.value}` };
    }
    // 情况 3：都没有 → 登录过期
    else {
        ElMessage.error('登录过期');
        router.push('/auth/login');
        throw new Error('No token');
    }

    let res = await fetch(url, options);

    // 401 自动刷新 + 重试
    if (res.status === 401 && tokenIsNeeded) {
        if (!refreshPromise) {
            refreshPromise = refreshTokenIfNeeded();  // 单例防并发
        }
        try {
            await refreshPromise;
            options.headers.Authorization = `Bearer ${token.value}`;
            return await fetch(url, options);  // 重试原请求
        } catch {
            ElMessage.error('登录已过期，请重新登录');
            router.push('/auth/login');
            throw new Error('Token refresh failed');
        } finally {
            refreshPromise = null;
        }
    }
    return res;
}
```

## 2. 多级缓存架构

搭建 Caffeine（L1，本地）+ Redis（L2，分布式）二级缓存，实现防击穿/穿透/雪崩三防与跨实例 L1 同步。

### 核心实现 — `cache/MultiLevelCache.java`

```java
@Component
public class MultiLevelCache {

    // ==================== 常量 ====================
    private static final int L1_TTL_SECONDS = 120;       // L1 Caffeine 过期时间
    private static final int L1_MAX_SIZE = 2000;          // L1 最大条目数
    private static final String NULL_MARKER = "__NULL__"; // 防穿透空值标记
    private static final String LOCK_PREFIX = "lock:";     // 防击穿锁前缀
    private static final long LOCK_TTL_SECONDS = 10;      // 锁过期时间
    private static final long RETRY_DELAY_MS = 50;        // 获取锁失败重试间隔
    private static final int MAX_RETRY = 3;               // 最大重试次数

    private final Cache<String, String> l1;               // Caffeine 本地缓存
    private final StringRedisTemplate redis;              // Redis 远程缓存

    public MultiLevelCache(StringRedisTemplate redis) {
        this.redis = redis;
        this.l1 = Caffeine.newBuilder()
                .maximumSize(L1_MAX_SIZE)                         // 最多 2000 条
                .expireAfterWrite(Duration.ofSeconds(L1_TTL_SECONDS)) // 120 秒过期
                .recordStats()                                     // 开启统计监控
                .build();
    }

    // ==================== 公共读取接口 ====================
    public <T> T get(String key, Class<T> clazz, long l2TtlSeconds, Supplier<T> dbLoader) {
        return getInternal(key, clazz, l2TtlSeconds, dbLoader, 0);
    }

    public <T> T get(String key, Type type, long l2TtlSeconds, Supplier<Object> dbLoader) {
        return getInternal(key, type, l2TtlSeconds, dbLoader, 0);
    }

    // ==================== 核心读取逻辑 ====================
    private <T> T getInternal(String key, Type type, long l2TtlSeconds,
                               Supplier<Object> dbLoader, int retryCount) {
        // ─── 第一层：L1 Caffeine ───
        String value = l1.getIfPresent(key);
        if (value != null) {
            return nullMarkerToNull(value, type);  // 命中 L1，直接返回
        }

        // ─── 第二层：L2 Redis ───
        String redisJson = redis.opsForValue().get(key);
        if (redisJson != null) {
            l1.put(key, redisJson);                // 回填 L1
            return nullMarkerToNull(redisJson, type);
        }

        // ─── 第三层：SETNX 分布式锁防击穿 ───
        String lockKey = LOCK_PREFIX + key;
        Boolean gotLock = redis.opsForValue()
                .setIfAbsent(lockKey, "1", Duration.ofSeconds(LOCK_TTL_SECONDS));

        if (Boolean.TRUE.equals(gotLock)) {
            try {
                // 双重检查 L2（获取锁后可能已被其他线程填充）
                redisJson = redis.opsForValue().get(key);
                if (redisJson != null) {
                    l1.put(key, redisJson);
                    return nullMarkerToNull(redisJson, type);
                }

                // 查询数据库
                Object data = dbLoader.get();

                // 防穿透：DB 返回 null 时写入 __NULL__ 标记
                String valueToSet = (data == null) ? NULL_MARKER : JSON.toJSONString(data);

                // 防雪崩：L2 TTL 附加 0~20% 随机抖动
                int jitter = ThreadLocalRandom.current().nextInt((int) (l2TtlSeconds * 0.2));
                int actualTtl = (int) l2TtlSeconds + jitter;

                // 回填 L2 + L1
                redis.opsForValue().set(key, valueToSet, Duration.ofSeconds(actualTtl));
                l1.put(key, valueToSet);

                return (T) data;
            } finally {
                redis.delete(lockKey);  // 释放锁
            }
        } else {
            // 未获取锁：重试等待
            if (retryCount >= MAX_RETRY) {
                log.warn("缓存重建重试耗尽 key={}，降级直接查DB", key);
                return (T) dbLoader.get();  // 降级：直接查 DB
            }
            try { Thread.sleep(RETRY_DELAY_MS); } catch (InterruptedException ignored) {}
            return getInternal(key, type, l2TtlSeconds, dbLoader, retryCount + 1);
        }
    }

    // ==================== 空值反序列化 ====================
    private <T> T nullMarkerToNull(String json, Type type) {
        if (NULL_MARKER.equals(json)) return null;
        return (T) JSON.parseObject(json, type);
    }

    // ==================== 驱逐（删除 L1 + L2 + 广播） ====================
    public void evict(String key) {
        l1.invalidate(key);
        redis.delete(key);
        redis.convertAndSend(CacheConfig.CACHE_EVICT_TOPIC, key);  // 广播给所有实例
    }

    public void evictByPrefix(String prefix) {
        // 清除 L1 中匹配前缀的 key
        l1.asMap().keySet().removeIf(k -> k.startsWith(prefix));
        // 清除 L2 中匹配前缀的 key
        Set<String> keys = redis.keys(prefix + "*");
        if (keys != null && !keys.isEmpty()) redis.delete(keys);
        // 广播通知其他实例
        redis.convertAndSend(CacheConfig.CACHE_EVICT_TOPIC, prefix);
    }

    // ==================== 跨实例同步接收端 ====================
    public void handleCacheEvict(String message) {
        if (message == null) return;
        l1.invalidate(message);
        // 支持前缀通配符：需要清除所有匹配的 L1 key
        if (message.contains("*")) {
            String prefix = message.replace("*", "");
            l1.asMap().keySet().removeIf(k -> k.startsWith(prefix));
        }
    }
}
```

### Redis Pub/Sub 配置 — `cache/CacheConfig.java`

```java
@Configuration
public class CacheConfig {

    public static final String CACHE_EVICT_TOPIC = "cache:evict";  // 广播频道

    @Bean
    public RedisTemplate<String, Object> redisTemplate(RedisConnectionFactory factory) {
        RedisTemplate<String, Object> template = new RedisTemplate<>();
        template.setConnectionFactory(factory);
        template.setKeySerializer(RedisSerializer.string());
        template.setValueSerializer(RedisSerializer.json());    // Jackson2 JSON 序列化
        return template;
    }

    // 消息监听容器：订阅 "cache:evict" 频道
    @Bean
    public RedisMessageListenerContainer cacheEvictContainer(
            RedisConnectionFactory factory, MessageListenerAdapter adapter) {
        RedisMessageListenerContainer container = new RedisMessageListenerContainer();
        container.setConnectionFactory(factory);
        container.addMessageListener(adapter, new ChannelTopic(CACHE_EVICT_TOPIC));
        return container;
    }

    // 消息适配器：收到消息 → 调用 MultiLevelCache.handleCacheEvict()
    @Bean
    public MessageListenerAdapter cacheEvictListener(MultiLevelCache cache) {
        return new MessageListenerAdapter(cache, "handleCacheEvict");
    }
}
```

## 4. WebSocket 实时聊天

基于 Jakarta WebSocket 实现即时通讯，通过 JWT query 参数完成握手认证，使用 `ConcurrentHashMap` 管理多 tab 会话映射，支持消息持久化、未读计数。

### WebSocket 服务端 — `webSocket/ChatWebSocket.java`

```java
@Slf4j
@Component
@ServerEndpoint("/websocket/auth/chat")   // 路径在 SecurityConfig + JwtFilter 白名单中
public class ChatWebSocket {

    // ==================== 多 Tab 会话管理 ====================
    // userId → Set<Session>: 一个用户多个浏览器 Tab 对应多个 WebSocket Session
    private static final ConcurrentHashMap<Integer, Set<Session>> SESSION_MAP
            = new ConcurrentHashMap<>();
    // Session → userId: 关闭时 O(1) 反向查找
    private static final ConcurrentHashMap<Session, Integer> USER_MAP
            = new ConcurrentHashMap<>();

    // 通过 SpringContextUtil 获取 Spring 管理的 Bean（WebSocket 是多实例的）
    private final JwtProvider jwtProvider = SpringContextUtil.getBean(JwtProvider.class);
    private final ChatService chatService = SpringContextUtil.getBean(ChatService.class);

    // ==================== 握手认证 ====================
    @OnOpen
    public void onOpen(Session session) {
        try {
            String token = extractToken(session.getQueryString());
            if (token == null || token.isEmpty()) {
                sendErrorAndClose(session, "Missing token parameter"); return;
            }

            Integer userId = jwtProvider.getUserId(token);  // JWT 解析
            if (userId == null) {
                sendErrorAndClose(session, "Invalid token"); return;
            }

            // 注册会话
            SESSION_MAP.computeIfAbsent(userId, k -> ConcurrentHashMap.newKeySet()).add(session);
            USER_MAP.put(session, userId);

            log.info("用户{} websocket 连接成功，当前该用户共{}个连接",
                    userId, SESSION_MAP.get(userId).size());
        } catch (Exception e) {
            log.error("websocket 连接失败", e);
            sendErrorAndClose(session, "Authentication failed");
        }
    }

    // ==================== 消息处理 ====================
    @OnMessage
    public void onMessage(String message, Session session) {
        // 心跳检测：客户端每 30 秒发 {"beat": 1}，服务端忽略
        if (message.contains("\"beat\"")) return;

        Integer userId = USER_MAP.get(session);
        ChatMessageDTO dto = JSON.parseObject(message, ChatMessageDTO.class);

        // ① 持久化消息到 MySQL
        ChatMsg msg = ChatMsg.builder()
                .senderUserId(userId)
                .targetUserId(dto.getTargetUserId())
                .content(dto.getContent())
                .isRead(0)
                .isAnonymous(dto.getIsAnonymous())
                .createdAt(LocalDateTime.now())
                .build();
        chatService.saveMessage(msg);

        // ② 推送消息给接收方（所有 Tab）
        sendToUser(dto.getTargetUserId(), JSON.toJSONString(msg));

        // ③ 同步到发送者自己的其他 Tab（多 Tab 消息同步）
        sendToUserExcept(userId, session, JSON.toJSONString(msg));
    }

    // ==================== 连接关闭 ====================
    @OnClose
    public void onClose(Session session) {
        Integer userId = USER_MAP.remove(session);
        if (userId != null) {
            SESSION_MAP.computeIfPresent(userId, (k, sessions) -> {
                sessions.remove(session);
                if (sessions.isEmpty()) {
                    log.info("用户{} websocket 全部断开", userId);
                    return null;  // 返回 null 移除整个 entry
                }
                log.info("用户{} websocket 断开一个连接，还剩{}个", userId, sessions.size());
                return sessions;
            });
        }
    }

    @OnError
    public void onError(Session session, Throwable error) {
        log.error("websocket 异常: userId={}", USER_MAP.get(session), error);
        onClose(session);
    }

    // ==================== 消息投递 ====================
    // 向指定用户的所有 Tab 发送消息
    private void sendToUser(Integer userId, String json) {
        Set<Session> sessions = SESSION_MAP.get(userId);
        if (sessions == null || sessions.isEmpty()) {
            log.warn("用户{}不在线，消息未送达", userId);
            return;
        }
        for (Session s : sessions) {
            if (s.isOpen()) {
                try { s.getBasicRemote().sendText(json); }
                catch (Exception e) { log.error("发送消息失败", e); }
            }
        }
    }

    // 向指定用户除 except 外的所有 Tab 发送（发消息者自己的其他 Tab 同步）
    private void sendToUserExcept(Integer userId, Session except, String json) {
        Set<Session> sessions = SESSION_MAP.get(userId);
        if (sessions != null) {
            for (Session s : sessions) {
                if (s != except && s.isOpen()) {
                    try { s.getBasicRemote().sendText(json); }
                    catch (Exception e) { log.error("同步消息失败", e); }
                }
            }
        }
    }

    // ==================== 工具方法 ====================
    // 从 URL query string 中提取 token
    private String extractToken(String queryString) {
        if (queryString == null) return null;
        try {
            for (String param : queryString.split("&")) {
                if (param.startsWith("token=")) {
                    return URLDecoder.decode(param.substring(6), StandardCharsets.UTF_8);
                }
            }
        } catch (Exception e) { log.error("解析 token 参数失败", e); }
        return null;
    }

    private void sendErrorAndClose(Session session, String msg) {
        try {
            session.getBasicRemote().sendText("{\"error\":\"" + msg + "\"}");
            session.close();
        } catch (Exception ignored) {}
    }
}
```

### 前端 WebSocket 客户端 — `composables/useWebsocket.ts`

```typescript
export function useWebSocket(url: string, options: WebSocketOptions = {}) {
    const { maxRetries = 5, reconnectDelay = 1000 } = options;

    let ws: WebSocket | null = null;
    let retryCount = 0;
    let heartbeatTimer: number | null = null;

    function connect() {
        ws = new WebSocket(url);

        ws.onopen = () => {
            retryCount = 0;
            startHeartbeat();  // 启动心跳
            options.onOpen?.();
        };

        ws.onmessage = (event) => {
            options.onMessage?.(event.data);
        };

        ws.onclose = () => {
            stopHeartbeat();
            if (retryCount < maxRetries) {
                const delay = reconnectDelay * Math.pow(2, retryCount);  // 指数退避
                retryCount++;
                setTimeout(connect, delay);  // 自动重连
            }
        };

        ws.onerror = () => {
            ws?.close();
        };
    }

    // 每 30 秒发送心跳
    function startHeartbeat() {
        heartbeatTimer = setInterval(() => {
            ws?.send(JSON.stringify({ beat: 1 }));
        }, 30000);
    }

    function send(data: object) {
        ws?.send(JSON.stringify(data));
    }

    function disconnect() {
        stopHeartbeat();
        ws?.close();
    }

    return { connect, send, disconnect };
}
```

```java
        Wrappers.lambdaQuery(Sue.class)
            .eq(Sue::getUserId, userId)
            .eq(Sue::getTargetId, targetID)
            .eq(Sue::getTargettype, targetType)
            .eq(Sue::getStatus, "wait"));     // 只检查待处理状态的举报
    if (count > 0) {
        throw new BizException("请勿重复举报");
    }

    // ─── 创建举报 ───
    Sue sue = new Sue();
    sue.setTargettype(targetType);
    sue.setTargetId(targetID);
    sue.setUserId(userId);
    sue.setReason(dto.getReason());
    sue.setTime(LocalDateTime.now());
    sue.setStatus("wait");     // 初始状态: 待处理
    sue.setFinish(false);
    sueMapper.insert(sue);
}
```

```java
    sue.setFinish(true);
    sueMapper.updateById(sue);

// 违规处理
public void violation(Integer sueID) {
    Sue sue = sueMapper.selectById(sueID);
    sue.setStatus("ok");
    sue.setFinish(true);
    sueMapper.updateById(sue);

    // ─── 定位违规者 ───
    Integer offenderId = null;
    switch (sue.getTargettype()) {
        case "post" -> {
            Post post = postMapper.selectById(sue.getTargetId());
            if (post != null) offenderId = post.getUserId();
        }
        case "pcomment" -> {
            Pcomment pc = pcommentMapper.selectById(sue.getTargetId());
            if (pc != null) offenderId = pc.getUserId();
        }
        case "ccomment" -> {
            Ccomment cc = ccommentMapper.selectById(sue.getTargetId());
            if (cc != null) offenderId = cc.getUserId();
        }
    }

    // ─── 累加违规次数 + 自动封禁 ───
    if (offenderId != null) {
        User offender = userMapper.selectById(offenderId);
        if (offender != null) {
            int newCount = (offender.getPunishNum() == null ? 0 : offender.getPunishNum()) + 1;
            offender.setPunishNum(newCount);
            offender.setUpdatedAt(LocalDateTime.now());

            // 累计 3 次违规 → 自动封禁 7 天
            if (newCount >= 3) {
                offender.setBanend(LocalDateTime.now().plusDays(7));
            }

            userMapper.updateById(offender);
        }
    }
}
```
